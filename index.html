PASSO 01

Abra o CodePen

Não precisa criar nenhuma pasta, arquivo ou instalar programa. Vamos fazer tudo diretamente pelo navegador.

1

Acesse o CodePen

Abra o CodePen e entre no editor de projetos.

2

Crie um Pen

No editor você encontrará três áreas: HTML, CSS e JS.

3

Copie os códigos

Copie cada código abaixo e cole na área correspondente.

Importante: não crie arquivos como index.html, style.css ou script.js. O CodePen já separa tudo para você.

PASSO 04

Coloque o JavaScript

Por último, copie este código e cole na área JS.

JavaScript 
let cliques = 0;


/* FUNÇÃO DO BOTÃO */

function comemorar() {

    cliques++;


    /* MENSAGEM */

    document.getElementById("mensagem").innerText =
        "Você fez seu site funcionar!";


    /* CONTADOR */

    document.getElementById("contador").innerText =
        "Cliques: " + cliques;


    /* BARRA DE PROGRESSO */

    let progresso = cliques * 10;

    if (progresso > 100) {
        progresso = 100;
    }

    document.getElementById("barra").style.width =
        progresso + "%";


    /* ANIMAÇÃO DO CARTÃO */

    let card = document.getElementById("card");

    card.classList.remove("pular");

    void card.offsetWidth;

    card.classList.add("pular");


    /* QUANTIDADE DE CONFETES */

    for (let i = 0; i < 40; i++) {

        let confete =
            document.createElement("div");


        confete.className = "confete";


        /* POSIÇÃO ALEATÓRIA */

        confete.style.left =
            Math.random() * 100 + "vw";


        /* TAMANHO ALEATÓRIO */

        let tamanho =
            Math.random() * 8 + 6;

        confete.style.width =
            tamanho + "px";

        confete.style.height =
            tamanho + "px";


        /* VELOCIDADE ALEATÓRIA */

        confete.style.setProperty(
            "--tempo",
            Math.random() * 2 + 2 + "s"
        );


        /* MOVIMENTO LATERAL */

        confete.style.setProperty(
            "--movimento",
            (Math.random() * 200 - 100) + "px"
        );


        /* CORES */

        confete.style.background =
            [
                "#315efb",
                "#ff4757",
                "#2ed573",
                "#ffa502",
                "#a55eea"
            ][
                Math.floor(Math.random() * 5)
            ];


        document.body.appendChild(confete);


        /* REMOVE O CONFETE */

        setTimeout(
            () => confete.remove(),
            4000
        );

    }

}
